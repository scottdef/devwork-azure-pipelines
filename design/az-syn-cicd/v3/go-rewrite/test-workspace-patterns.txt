# test-workspace-patterns.txt
# Maps dev workspace naming patterns to test workspace equivalents.
#
# Format: match=replace (first = is the delimiter)
# Lines starting with # are comments.  Empty lines are ignored.
#
# Order matters: patterns are applied sequentially from top to bottom.
# Place longer, more specific patterns before shorter, broader ones
# to avoid partial matches corrupting longer names.
# 
# Example: "dev-sql-server01" should be listed before "dev" to prevent
# the shorter pattern from rewriting it to "tst-sql-server01" before
# the specific pattern can match.

# ── Workspace name ──────────────────────────────────────────────────────
synapse-workspace-dev=synapse-workspace-test

# ── Resource group ──────────────────────────────────────────────────────
rg-synapse-dev=rg-synapse-test

# ── Linked services (specific names first, then generic patterns) ──────
link-svc-dev-db-sql01=link-svc-tst-db-sql01
link-svc-dev-db-sql02=link-svc-tst-db-sql02
link-svc-dev-kv=link-svc-tst-kv

link-svc-dev-adls=link-svc-tst-adls

# ── Storage accounts ───────────────────────────────────────────────────

# ── Key Vaults ─────────────────────────────────────────────────────────
kv-synapse-dev=kv-synapse-test
