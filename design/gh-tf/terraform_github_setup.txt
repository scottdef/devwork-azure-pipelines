  description = "Azure tenant ID"

  description = "Databricks workspace URL"
  type        = string

# Create the repository
resource "github_repository" "main" {
  name         = var.repository_name
  description  = "Databricks Asset Bundle deployment with Azure Document Intelligence"
  visibility   = "private"
  
  has_issues   = true
  has_wiki     = true
  has_projects = true
  
  auto_init          = true
  gitignore_template = "Python"
  license_template   = "mit"
  
  allow_merge_commit     = true
  allow_squash_merge     = true
  allow_rebase_merge     = true
  delete_branch_on_merge = true
  
  security_and_analysis {
    secret_scanning {
      status = "enabled"
    }
    secret_scanning_push_protection {
      status = "enabled"
    }
  }
}

# Create team
resource "github_team" "team_alpha_blue_i" {
  name        = "team-alpha-blue-i"
  description = "Alpha Blue I Development Team"
  privacy     = "closed"
}

# Add team to repository with admin access
resource "github_team_repository" "team_access" {
  team_id    = github_team.team_alpha_blue_i.id
  repository = github_repository.main.name
  permission = "admin"
}

# Branch protection rule
resource "github_branch_protection" "main" {
  repository_id = github_repository.main.node_id
  pattern       = "main"
  
  required_status_checks {
    strict = true
    contexts = [
      "ci/databricks-validate",
      "ci/terraform-validate"
    ]
  }
  
  required_pull_request_reviews {
    required_approving_review_count = 2
    require_code_owner_reviews      = true
    dismiss_stale_reviews           = true
    restrict_dismissals             = false
  }
  
  restrict_pushes {
    blocks_creations = false
  }
  
  enforce_admins = false
  allows_deletions = false
  allows_force_pushes = false
}

# Environment variables for nonprod
resource "github_actions_environment_variable" "nonprod_vars" {
  for_each = {
    AZURE_SUBSCRIPTION_ID      = var.azure_subscription_id
    AZURE_TENANT_ID           = var.azure_tenant_id
    DATABRICKS_WORKSPACE_URL  = var.databricks_workspace_url
    ENVIRONMENT              = "nonprod"
    AZURE_RESOURCE_GROUP     = "rg-databricks-nonprod"
    DOCUMENT_INTELLIGENCE_ENDPOINT = "https://doc-intel-nonprod.cognitiveservices.azure.com/"
  }
  
  repository    = github_repository.main.name
  environment   = github_repository_environment.nonprod.environment
  variable_name = each.key
  value         = each.value
}
