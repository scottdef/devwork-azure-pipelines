# CI/CD Pipeline Template
# Created via IssueOps - Replace with your actual workflow

name: CI/CD Pipeline

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pull-requests: write

jobs:
  # Build and test
  build:
    name: Build & Test
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Setup Environment
        run: |
          echo "Configure your build environment here"
          # Examples:
          # - Setup Node: uses: actions/setup-node@v4
          # - Setup Python: uses: actions/setup-python@v4
          # - Setup Go: uses: actions/setup-go@v4

      - name: Install Dependencies
        run: |
          echo "Install your dependencies here"
          # Examples:
          # - npm install
          # - pip install -r requirements.txt
          # - go mod download

      - name: Run Linting
        run: |
          echo "Run your linters here"
          # Examples:
          # - npm run lint
          # - flake8 .
          # - golangci-lint run

      - name: Run Tests
        run: |
          echo "Run your tests here"
          # Examples:
          # - npm test
          # - pytest
          # - go test ./...

      - name: Build Application
        run: |
          echo "Build your application here"
          # Examples:
          # - npm run build
          # - python setup.py build
          # - go build -o app ./cmd/main.go

  # Security scanning
  security:
    name: Security Scan
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Run Trivy Scanner
        uses: aquasecurity/trivy-action@master
        with:
          scan-type: 'fs'
          scan-ref: '.'
          format: 'sarif'
          output: 'trivy-results.sarif'

      - name: Upload Results
        uses: github/codeql-action/upload-sarif@v3
        with:
          sarif_file: 'trivy-results.sarif'

  # Deploy (only on main branch)
  deploy:
    name: Deploy
    runs-on: ubuntu-latest
    needs: [build, security]
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Deploy Application
        run: |
          echo "Deploy your application here"
          # Examples:
          # - Deploy to Kubernetes
          # - Deploy to AWS/GCP/Azure
          # - Deploy to Heroku
          # - Push Docker image

      - name: Create Deployment Record
        uses: actions/github-script@v7
        with:
          script: |
            await github.rest.repos.createDeployment({
              owner: context.repo.owner,
              repo: context.repo.repo,
              ref: context.sha,
              environment: 'production',
              auto_merge: false,
              required_contexts: []
            });
