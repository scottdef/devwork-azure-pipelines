# Makefile for HEREDOC Workflow Management
# Rob Pike would approve of this simplicity

.PHONY: help generate-local generate-remote generate-cross test-heredoc clean-generated list-workflows

GREEN := \033[0;32m
BLUE := \033[0;34m
YELLOW := \033[1;33m
NC := \033[0m

help: ## Show this help
	@echo ""
	@echo "$(BLUE)HEREDOC Workflow Targets:$(NC)"
	@echo ""
	@grep -E '^[a-zA-Z_-]+:.*?## .*$$' $(MAKEFILE_LIST) | \
		awk 'BEGIN {FS = ":.*?## "}; {printf "  $(GREEN)%-25s$(NC) %s\n", $$1, $$2}'
	@echo ""

generate-local: ## Generate files locally (test before workflow)
	@echo "$(BLUE)Generating files locally...$(NC)"
	@mkdir -p here-files
	@cat > here-files/test-file.txt <<-'EOF'
		# Test File
		Generated locally at: $$(date)
		This is a test of HEREDOC file generation.
	EOF
	@cat > here-files/test-script.sh <<-'EOF'
		#!/bin/bash
		echo "This is a test script"
		echo "Generated: $$(date)"
	EOF
	@chmod +x here-files/test-script.sh
	@echo "$(GREEN)✓ Files generated in here-files/$(NC)"
	@ls -lh here-files/

generate-remote: ## Trigger remote generation workflow
	@if [ -z "$(TARGET_REPO)" ]; then \
		echo "$(YELLOW)Usage: make generate-remote TARGET_REPO=owner/repo$(NC)"; \
		exit 1; \
	fi
	@echo "$(BLUE)Triggering workflow for $(TARGET_REPO)...$(NC)"
	@gh workflow run generate-files.yml \
		-f target_repo="$(TARGET_REPO)" \
		-f branch_name="$${BRANCH:-generated-files}" \
		-f pr_title="$${TITLE:-Add generated files}"
	@echo "$(GREEN)✓ Workflow triggered$(NC)"
	@echo "Check: https://github.com/$(shell gh repo view --json nameWithOwner -q .nameWithOwner)/actions"

generate-cross: ## Trigger cross-repo generation
	@if [ -z "$(OWNER)" ] || [ -z "$(REPO)" ]; then \
		echo "$(YELLOW)Usage: make generate-cross OWNER=org REPO=repo$(NC)"; \
		exit 1; \
	fi
	@echo "$(BLUE)Triggering cross-repo generation...$(NC)"
	@gh workflow run cross-repo-generate.yml \
		-f owner="$(OWNER)" \
		-f repo="$(REPO)" \
		-f branch="$${BRANCH:-auto-generated-files}"
	@echo "$(GREEN)✓ Workflow triggered$(NC)"

generate-simple: ## Trigger simple generation (current repo)
	@echo "$(BLUE)Triggering simple generation...$(NC)"
	@gh workflow run generate-files-simple.yml
	@echo "$(GREEN)✓ Workflow triggered$(NC)"

test-heredoc: ## Test HEREDOC syntax locally
	@echo "$(BLUE)Testing HEREDOC syntax...$(NC)"
	@echo ""
	@echo "Test 1: Basic HEREDOC"
	@cat <<-'EOF'
		This is a basic HEREDOC test.
		No variable expansion.
	EOF
	@echo ""
	@echo "Test 2: HEREDOC with expansion"
	@cat <<-EOF
		Current directory: $$(pwd)
		User: $$USER
		Date: $$(date)
	EOF
	@echo ""
	@echo "Test 3: JSON generation"
	@cat <<-'EOF' | jq empty && echo "$(GREEN)✓ Valid JSON$(NC)" || echo "$(YELLOW)✗ Invalid JSON$(NC)"
		{
		  "test": true,
		  "value": 123
		}
	EOF
	@echo ""

list-workflows: ## List all workflow runs
	@echo "$(BLUE)Recent workflow runs:$(NC)"
	@gh run list --limit 10

watch-workflows: ## Watch workflow runs (live)
	@echo "$(BLUE)Watching workflows... (Ctrl+C to stop)$(NC)"
	@gh run watch

view-workflow: ## View specific workflow run
	@if [ -z "$(RUN_ID)" ]; then \
		echo "$(YELLOW)Usage: make view-workflow RUN_ID=123456$(NC)"; \
		gh run list --limit 5; \
		exit 1; \
	fi
	@gh run view $(RUN_ID) --log

clean-generated: ## Clean generated files locally
	@echo "$(BLUE)Cleaning generated files...$(NC)"
	@rm -rf here-files/
	@echo "$(GREEN)✓ Cleaned$(NC)"

verify-setup: ## Verify workflow setup
	@echo "$(BLUE)Verifying setup...$(NC)"
	@echo ""
	@echo "Checking gh CLI..."
	@command -v gh >/dev/null 2>&1 && echo "$(GREEN)✓ gh CLI installed$(NC)" || echo "$(YELLOW)✗ gh CLI not found$(NC)"
	@echo ""
	@echo "Checking authentication..."
	@gh auth status >/dev/null 2>&1 && echo "$(GREEN)✓ Authenticated$(NC)" || echo "$(YELLOW)✗ Not authenticated$(NC)"
	@echo ""
	@echo "Checking workflows..."
	@[ -f .github/workflows/generate-files.yml ] && echo "$(GREEN)✓ generate-files.yml$(NC)" || echo "$(YELLOW)✗ Missing$(NC)"
	@[ -f .github/workflows/generate-files-simple.yml ] && echo "$(GREEN)✓ generate-files-simple.yml$(NC)" || echo "$(YELLOW)✗ Missing$(NC)"
	@[ -f .github/workflows/cross-repo-generate.yml ] && echo "$(GREEN)✓ cross-repo-generate.yml$(NC)" || echo "$(YELLOW)✗ Missing$(NC)"
	@echo ""

create-test-repo: ## Create test repository for testing
	@if [ -z "$(REPO_NAME)" ]; then \
		echo "$(YELLOW)Usage: make create-test-repo REPO_NAME=test-heredoc$(NC)"; \
		exit 1; \
	fi
	@echo "$(BLUE)Creating test repository: $(REPO_NAME)...$(NC)"
	@gh repo create $(REPO_NAME) --private --clone
	@cd $(REPO_NAME) && \
		echo "# Test Repository" > README.md && \
		git add README.md && \
		git commit -m "Initial commit" && \
		git push -u origin main
	@echo "$(GREEN)✓ Test repository created$(NC)"
	@echo "Now run: make generate-remote TARGET_REPO=$(shell gh api user | jq -r .login)/$(REPO_NAME)"

# Advanced targets

batch-generate: ## Generate files for multiple repos
	@if [ -z "$(REPOS)" ]; then \
		echo "$(YELLOW)Usage: make batch-generate REPOS='org/repo1 org/repo2'$(NC)"; \
		exit 1; \
	fi
	@for repo in $(REPOS); do \
		echo "$(BLUE)Triggering for $$repo...$(NC)"; \
		gh workflow run generate-files.yml -f target_repo="$$repo"; \
		sleep 2; \
	done
	@echo "$(GREEN)✓ All workflows triggered$(NC)"

approve-prs: ## List PRs created by workflows
	@echo "$(BLUE)PRs created by github-actions[bot]:$(NC)"
	@gh pr list --author app/github-actions --json number,title,url \
		--template '{{range .}}#{{.number}} - {{.title}}
{{.url}}
{{end}}'

merge-auto-prs: ## Auto-merge PRs from workflows (use carefully!)
	@echo "$(YELLOW)⚠ This will merge all PRs from github-actions[bot]$(NC)"
	@read -p "Are you sure? (yes/no): " confirm; \
	if [ "$$confirm" = "yes" ]; then \
		gh pr list --author app/github-actions --json number \
			--jq '.[].number' | \
		while read -r pr; do \
			echo "Merging PR #$$pr..."; \
			gh pr merge $$pr --squash --delete-branch; \
		done; \
		echo "$(GREEN)✓ PRs merged$(NC)"; \
	else \
		echo "Cancelled"; \
	fi
